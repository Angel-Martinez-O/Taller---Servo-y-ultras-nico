#include <Servo.h>

const int trig = 7;
const int echo = 6;
const int pinServo = 9;

const int posCerrada = 0;
const int posMedia = 45;
const int posAbierta = 90;

Servo puerta;

void setup() {
  Serial.begin(9600);
  pinMode(trig, OUTPUT);
  pinMode(echo, INPUT);
  puerta.attach(pinServo);
  puerta.write(posCerrada);
}

void loop() {
  digitalWrite(trig, LOW);
  delayMicroseconds(2);
  digitalWrite(trig, HIGH);
  delayMicroseconds(10);
  digitalWrite(trig, LOW);

  long tiempo = pulseIn(echo, HIGH);
  float distancia = tiempo * 0.034 / 2;

  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.println(" cm");

  if (distancia > 30) {
    puerta.write(posCerrada);
    Serial.println("PUERTA CERRADA (0°)");
  } else if (distancia > 15) {
    puerta.write(posMedia);
    Serial.println("PUERTA A MEDIO ABRIR (45°)");
  } else {
    puerta.write(posAbierta);
    Serial.println("PUERTA ABIERTA (90°)");
  }

  delay(100);
}
